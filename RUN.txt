本地启动：

安装go:

# 下载 Go 二进制（替换为最新版，如 go1.22.5）
wget https://go.dev/dl/go1.22.5.linux-amd64.tar.gz

# 解压到 /usr/local
sudo tar -C /usr/local -xzf go1.22.5.linux-amd64.tar.gz

# 将 Go 加入 PATH（永久生效）
echo 'export PATH=$PATH:/usr/local/go/bin' >> ~/.bashrc
source ~/.bashrc

# 验证安装
go version




------------------------------------------------------------------------------------------

# 1. 启动服务


# 设置好权限
sudo usermod -aG docker $USER
newgrp docker
docker ps

# 中国可能需要设置仓库地址
sudo vim /etc/docker/daemon.json

# 输入
{
  "registry-mirrors": [
    "https://docker.m.daocloud.io",
    "https://docker.1ms.run",
    "https://hub.rat.dev"
  ]
}

# 重启服务
sudo systemctl daemon-reload
sudo systemctl restart docker

# 验证配置是否生效
docker info | grep -A 1 "Registry Mirrors"

#  利用docker配置文件下载相关依赖
cd /path/to/your/docker-compose-dir
docker-compose up -d

# 2. 创建 Kafka topic
docker exec -it go-data-bridge_kafka_1  kafka-topics.sh --create --bootstrap-server localhost:9092 --topic my_topic --partitions 1 --replication-factor 1

# 3. 创建 ClickHouse 表
docker exec -it go-data-bridge_clickhouse_1 clickhouse-client -q "CREATE TABLE IF NOT EXISTS my_table (data String) ENGINE = MergeTree() ORDER BY tuple();"

# 4. 修改 Go 配置文件，确保 brokers 和 hosts 是 localhost:9092 / localhost:9000

# 5. 运行 Go 程序
cd /path/to/your/go/go-data-bridge
go run main.go

# 6. 打开另一个终端，发送测试消息
docker exec -it go-data-bridge_kafka_1  kafka-console-producer.sh --broker-list localhost:9092 --topic my_topic
# 输入消息
